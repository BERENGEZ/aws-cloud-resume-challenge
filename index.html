Infrastructure Configurations

server {
    listen 80;
    server_name barakaberenge.com;
    
    location / { root /var/www/portfolio; index index.html; }
    location /api/ { proxy_pass http://127.0.0.1:5000; }
}
[Service]
User=www-data
WorkingDirectory=/var/www/portfolio/api
ExecStart=/var/www/portfolio/api/venv/bin/gunicorn --workers 3 --bind 127.0.0.1:5000 app:app
Restart=always
